# Equations for HCM with random coefficients

$$
\text{Utility functions}\\
\begin{aligned}
V_{\mathrm{mv},n,t} =
&\sum_{h=1}^3 \big( \beta_{\mathrm{hosp}[h]} + \beta_{\mathrm{hosp*risk}}x_{\mathrm{risk},n,t} \big) x_{\mathrm{hosp}[h],n,t}\ +\\
&\beta_{\mathrm{cost},n}x_{\mathrm{cost},n,t} + \beta_{\mathrm{redness}}x_{\mathrm{redness},n,t}\ +\\ 
&\beta_{\mathrm{preterm},n}x_{\mathrm{preterm},n,t} + \beta_{\mathrm{pregnant},n}x_{\mathrm{pregnant},n,t}\\
\end{aligned}\\[0.4em]

\begin{aligned}
V_{\mathrm{mab},n,t} =
&\sum_{h=1}^3 \big( \beta_{\mathrm{hosp}[h]} + \beta_{\mathrm{hosp*risk}}x_{\mathrm{risk},n,t} \big) x_{\mathrm{hosp}[h],n,t}\ +\\
&\beta_{\mathrm{cost},n}x_{\mathrm{cost},n,t} + \beta_{\mathrm{redness}}x_{\mathrm{redness},n,t}\\
\end{aligned}\\[0.4em]

V_{\mathrm{opt},n,t} = \beta_{\mathrm{opt*LV}}\lambda_n  + \beta_{\mathrm{opt*risk}}x_{\mathrm{risk},n,t} + \alpha_{\mathrm{opt},n}
$$

$$
\text{Linear predictor for latent variable}\\
\begin{aligned}
\lambda_n =\ 
&\gamma_{\mathrm{age}}\tilde{z}_{\mathrm{age},n} +
\gamma_{\mathrm{sibling}}z_{\mathrm{sibling},n}\ +\\
&\gamma_{\mathrm{education}}z_{\mathrm{education},n} +
\gamma_{\mathrm{income}}z_{\mathrm{income},n} + \eta_n
\end{aligned}
$$

$$
\text{Random coefficients}\\
\beta_{\mathrm{cost},n}
\sim \mathrm{Normal}(\mu_{\mathrm{cost}}, \sigma_{\mathrm{cost}})\\
\beta_{\mathrm{preterm},n}
\sim \mathrm{Normal}(\mu_{\mathrm{preterm}}, \sigma_{\mathrm{preterm}})\\
\beta_{\mathrm{pregnant},n} 
\sim \mathrm{Normal}(\mu_{\mathrm{pregnant}}, \sigma_{\mathrm{pregnant}})\\
\alpha_{\mathrm{opt},n} 
\sim \mathrm{Normal}(0, \sigma_{\mathrm{opt}})\\
\eta_n \sim \mathrm{Normal}(0,1)
$$

$$
\text{Likelihoods}\\
L_{C[n]}(\beta, \alpha, \lambda) = 
\prod_{t=1}^9 \frac{\exp(V_{j*,n,t})}{\sum_{j=1}^3 \exp(V_{j,n,t})} \\
L_{I[n]}(\tau, \zeta, \lambda) = 
\prod_{i=1}^5 \Big( \sum_{l=1}^4 \mathbb{1}_{(I_{i,n}=l)} \big[ P(I_{i,n} = l| \lambda_n) \big] \Big) \text{, where}\\
P(I_{i,n}=l|\lambda_n) = 
\operatorname{logit}^{-1}(\tau_{i,l}-\zeta_i\lambda_n) - \operatorname{logit}^{-1}(\tau_{i,l-1}-\zeta_i\lambda_n)\\
LL(\Omega) = 
\sum_{n=1}^N \log \int L_{C[n]}(\beta, \alpha, \lambda) L_{I[n]}(\tau, \zeta, \lambda) f(\beta_n) g(\alpha_n) h(\eta_n) d\beta_n d\alpha_n d\eta_n
$$
